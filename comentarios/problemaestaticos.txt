from .models import Vendas
from .forms import Vendaforms, ProdutoFormSet

def editar_venda(request, pk):
    venda = Vendas.objects.get(pk=pk)
    form = Vendaforms(request.POST or None, instance=venda)
    formset = ProdutoFormSet(request.POST or None, instance=venda)

    if request.method == 'POST':
        if form.is_valid() and formset.is_valid():
            form.save()
            formset.save()
            return redirect('detalhes_venda', pk=venda.pk)

    return render(request, 'vendas/editar_venda.html', {
        'form': form,
        'formset': formset,
        'venda': venda
    })
